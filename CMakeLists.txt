cmake_minimum_required(VERSION 3.14)
project(mini_redis VERSION 1.0.0 LANGUAGES CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "mini-redis requires Linux. Use WSL2 or Docker on Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BUILD_TESTING "Build unit and integration tests" ON)
option(BUILD_BENCHMARKS "Fetch Google Benchmark and build microbenchmarks" OFF)
option(BUILD_FUZZER "Build libFuzzer targets (Clang)" OFF)
option(ENABLE_COVERAGE "Enable gcov instrumentation" OFF)
set(SANITIZE "" CACHE STRING "address, undefined, address,undefined, or thread")
add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wconversion)
if(SANITIZE)
  add_compile_options(-fsanitize=${SANITIZE} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${SANITIZE})
endif()
if(ENABLE_COVERAGE)
  # Worker threads must not race on GCC's instrumentation counters.
  add_compile_options(--coverage -O0 -g -fprofile-update=atomic)
  add_link_options(--coverage)
endif()
find_package(Threads REQUIRED)
add_library(mini_redis_lib
  src/core/store.cpp src/core/lru.cpp src/core/expiry.cpp
  src/protocol/resp_parser.cpp src/protocol/resp_writer.cpp src/protocol/cli_formatter.cpp
  src/commands/command_table.cpp
  src/persistence/crc32.cpp src/persistence/snapshot.cpp src/persistence/aof.cpp
  src/config/config.cpp src/util/string_utils.cpp src/util/logger.cpp
  src/net/socket_util.cpp src/net/thread_pool.cpp src/net/server.cpp)
target_include_directories(mini_redis_lib PUBLIC src)
target_link_libraries(mini_redis_lib PUBLIC Threads::Threads)
add_executable(mini-redis src/main.cpp)
target_link_libraries(mini-redis PRIVATE mini_redis_lib)
add_executable(load_gen bench/load_gen.cpp)
target_link_libraries(load_gen PRIVATE mini_redis_lib)
include(FetchContent)
if(BUILD_TESTING)
  enable_testing()
  FetchContent_Declare(googletest URL https://codeload.github.com/google/googletest/tar.gz/refs/tags/v1.15.2)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
  add_executable(unit_tests tests/unit/core_test.cpp tests/unit/protocol_test.cpp tests/unit/persistence_test.cpp tests/unit/commands_test.cpp)
  target_link_libraries(unit_tests PRIVATE mini_redis_lib GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(unit_tests DISCOVERY_TIMEOUT 60)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/test_server.py $<TARGET_FILE:mini-redis>)
  set_tests_properties(integration PROPERTIES TIMEOUT 180)
  add_test(NAME dashboard COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/test_dashboard.py $<TARGET_FILE:mini-redis>)
  set_tests_properties(dashboard PROPERTIES TIMEOUT 60)
endif()
if(BUILD_BENCHMARKS)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(googlebenchmark URL https://codeload.github.com/google/benchmark/tar.gz/refs/tags/v1.9.1)
  FetchContent_MakeAvailable(googlebenchmark)
  add_executable(micro_store_bench bench/micro_store_bench.cpp)
  target_link_libraries(micro_store_bench PRIVATE mini_redis_lib benchmark::benchmark_main)
endif()
if(BUILD_FUZZER)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "BUILD_FUZZER requires Clang")
  endif()
  # Instrument the parser itself, not just the harness.
  foreach(target resp_parser_fuzz aof_fuzz)
    add_executable(${target} tests/fuzz/${target}.cpp src/protocol/resp_parser.cpp src/util/string_utils.cpp)
    target_include_directories(${target} PRIVATE src)
    target_compile_options(${target} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(${target} PRIVATE -fsanitize=fuzzer,address,undefined)
  endforeach()
  target_sources(aof_fuzz PRIVATE src/persistence/aof.cpp src/net/socket_util.cpp src/protocol/resp_writer.cpp)
endif()
